### SQLite schema (.tables + .schema) ###
inference_records  sessions         

CREATE TABLE sessions (
	id VARCHAR NOT NULL, 
	created_at DATETIME, 
	description VARCHAR, 
	PRIMARY KEY (id)
);
CREATE TABLE inference_records (
	id INTEGER NOT NULL, 
	session_id VARCHAR, 
	window_id INTEGER, 
	timestamp VARCHAR, 
	p_distraction FLOAT, 
	p_telemetry_anomaly FLOAT, 
	risk_score FLOAT, 
	risk_level VARCHAR, 
	dominant_evidence VARCHAR, 
	alert_severity VARCHAR, 
	system_health VARCHAR, 
	latency_ms INTEGER, 
	flagged BOOLEAN, 
	payload JSON, 
	PRIMARY KEY (id), 
	FOREIGN KEY(session_id) REFERENCES sessions (id) ON DELETE CASCADE
);
CREATE INDEX ix_inference_records_window_id ON inference_records (window_id);
CREATE INDEX ix_inference_records_session_id ON inference_records (session_id);
